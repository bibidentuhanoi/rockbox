# ── lang_enum.h + lang_core.c: generated by genlang from english.lang ─────────
# Feature string generated by preprocessing features.txt with the full config
# chain (mirrors apps/apps.make: cpp -E -P -imacros config.h features.txt | tr '\n' ':')
# This picks up HAVE_* defines from config.h (MEMORYSIZE > 2 → crossfade, pitchscreen, etc.)
# as well as direct defines in esp32.h — no manual maintenance needed.

function(rockbox_get_features RESULT_VAR)
    execute_process(
        COMMAND ${CMAKE_C_COMPILER}
            -E -P -x c
            -include ${BUILDDIR}/autoconf.h
            -include ${FIRMDIR}/export/config.h
            ${RB_PP_INCLUDES}
            ${RB_PP_DEFS}
            ${APPSDIR}/features.txt
        OUTPUT_VARIABLE _raw
        ERROR_VARIABLE  _err
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
    string(REPLACE "\n" ";" _lines "${_raw}")
    set(_features "")
    foreach(_line ${_lines})
        string(STRIP "${_line}" _line)
        if("${_line}" STREQUAL "" OR _line MATCHES "^#")
            continue()
        endif()
        if(_features)
            set(_features "${_features}:${_line}")
        else()
            set(_features "${_line}")
        endif()
    endforeach()
    set(${RESULT_VAR} "${_features}" PARENT_SCOPE)
endfunction()

# ESP-IDF evaluates component CMakeLists.txt twice:
#   Pass 1 (component_get_requirements): CMAKE_C_COMPILER not set — skip genlang
#   Pass 2 (actual configure): compiler available — run genlang
if(CMAKE_C_COMPILER)
    rockbox_get_features(_rb_features)
    set(_rb_targets "${_rb_features}")
    message(STATUS "Rockbox genlang features: ${_rb_targets}")

    file(MAKE_DIRECTORY ${BUILDDIR}/lang)
    execute_process(
        COMMAND perl ${RB_DIR}/tools/genlang
            -p=${BUILDDIR}/lang
            -t=${_rb_targets}
            -e=${APPSDIR}/lang/english.lang
            ${APPSDIR}/lang/english.lang
        RESULT_VARIABLE _genlang_result
        ERROR_VARIABLE  _genlang_err
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
    if(_genlang_result)
        message(FATAL_ERROR "genlang failed: ${_genlang_err}")
    endif()
    # genlang writes the LANG_ enum to ${BUILDDIR}/lang_enum.h (prefix_enum.h)
    # and the string table to ${BUILDDIR}/lang/lang_core.c
    configure_file(${BUILDDIR}/lang/lang_core.c ${BUILDDIR}/lang_core.c COPYONLY)

    # ── Bitmap headers: build bmp2rb host tool, generate .h + .c from .bmp ──────
    # Mirrors apps/bitmaps/bitmaps.make:
    #   BMP2RB_MONO   = bmp2rb -f 0  (1-bit mono)
    #   BMP2RB_NATIVE = bmp2rb -f 4  (16-bit RGB565, for ESP32 320x240)
    find_program(HOST_CC NAMES cc gcc REQUIRED)
    set(_bmp2rb_src ${RB_DIR}/tools/bmp2rb.c)
    set(_bmp2rb_bin ${BUILDDIR}/tools/bmp2rb)
    file(MAKE_DIRECTORY ${BUILDDIR}/tools ${BUILDDIR}/bitmaps)
    execute_process(
        COMMAND ${HOST_CC} -DAPPLICATION_NAME="bmp2rb" -o ${_bmp2rb_bin} ${_bmp2rb_src} -lm
        RESULT_VARIABLE _bmp2rb_cc_result
        ERROR_VARIABLE  _bmp2rb_cc_err
    )
    if(_bmp2rb_cc_result)
        message(FATAL_ERROR "Failed to compile bmp2rb: ${_bmp2rb_cc_err}")
    endif()

    function(run_bmp2rb FORMAT SRC_BMP OUT_C)
        execute_process(
            COMMAND ${_bmp2rb_bin} -f ${FORMAT} -b -h ${BUILDDIR}/bitmaps ${SRC_BMP}
            OUTPUT_FILE ${OUT_C}
            RESULT_VARIABLE _r
            ERROR_VARIABLE  _e
        )
        if(_r)
            message(FATAL_ERROR "bmp2rb failed on ${SRC_BMP}: ${_e}")
        endif()
    endfunction()

    # mono/default_icons  →  -f 0
    run_bmp2rb(0
        ${APPSDIR}/bitmaps/mono/default_icons.bmp
        ${BUILDDIR}/bitmaps/default_icons.c)

    # native/rockboxlogo (240x320 portrait, 16-bit)  →  -f 4
    # Rockbox selects logo by LCD_WIDTH: 240 → rockboxlogo.240x74x16.bmp (see apps/bitmaps/native/SOURCES)
    # TODO: if you ever switch back to landscape 320×240, change back to rockboxlogo.320x98x16.bmp
    run_bmp2rb(4
        ${APPSDIR}/bitmaps/native/rockboxlogo.240x74x16.bmp
        ${BUILDDIR}/bitmaps/rockboxlogo.240x74x16.c)

    # native/usblogo (LCD_WIDTH > 176, color)  →  -f 4
    run_bmp2rb(4
        ${APPSDIR}/bitmaps/native/usblogo.176x48x16.bmp
        ${BUILDDIR}/bitmaps/usblogo.176x48x16.c)

else()
    message(STATUS "Rockbox: skipping genlang in requirements phase (pass 1)")
endif()

# ── CPP-in-CMake: replicate Rockbox's preprocess() from functions.make ───────
# $(CC) -E -P -x c -include config.h SOURCES | grep/sed
function(rockbox_get_sources SOURCES_FILE BASE_DIR RESULT_VAR)
    execute_process(
        COMMAND ${CMAKE_C_COMPILER}
            -E -P -x c
            -include ${BUILDDIR}/autoconf.h
            -include ${FIRMDIR}/export/config.h
            ${RB_PP_INCLUDES}
            ${RB_PP_DEFS}
            ${SOURCES_FILE}
        OUTPUT_VARIABLE _raw
        ERROR_VARIABLE  _err
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )

    # Split on newlines and build file list
    string(REPLACE "\n" ";" _lines "${_raw}")
    set(_result "")
    foreach(_line ${_lines})
        string(STRIP "${_line}" _line)
        # Skip blank and CPP comment lines
        if("${_line}" STREQUAL "" OR _line MATCHES "^#")
            continue()
        endif()
        set(_full "${BASE_DIR}/${_line}")
        if(EXISTS "${_full}")
            list(APPEND _result "${_full}")
        endif()
    endforeach()
    set(${RESULT_VAR} ${_result} PARENT_SCOPE)
endfunction()

# ── Minimal codeclib.h shim for voice speex build ────────────────────────────
# config-speex.h includes codeclib.h for ICODE_ATTR etc.  The real codeclib.h
# lives in lib/rbcodec/codecs/lib/ and drags in the full codec API (platform.h,
# codecs.h, mdct.h, fft.h) plus redefines malloc/free — unusable for firmware-
# embedded voice speex.  On hosted platforms these attrs are all empty, so a
# tiny shim is sufficient.
file(MAKE_DIRECTORY ${BUILDDIR}/speex-shim)
file(WRITE ${BUILDDIR}/speex-shim/codeclib.h
"#ifndef __CODECLIB_H__\n"
"#define __CODECLIB_H__\n"
"#include \"config.h\"\n"
"#include <stdlib.h>\n"
"#include <string.h>\n"
"/* Voice-codec shim: section attributes (empty on hosted) */\n"
"#ifndef ICODE_ATTR\n"
"#define ICODE_ATTR\n"
"#endif\n"
"#ifndef IDATA_ATTR\n"
"#define IDATA_ATTR\n"
"#endif\n"
"#ifndef IBSS_ATTR\n"
"#define IBSS_ATTR\n"
"#endif\n"
"#ifndef ICONST_ATTR\n"
"#define ICONST_ATTR\n"
"#endif\n"
"/* Allocation functions — voice speex uses real libc on ESP32 */\n"
"#define codec_malloc  malloc\n"
"#define codec_calloc  calloc\n"
"#define codec_realloc realloc\n"
"#define codec_free    free\n"
"#define codec_strlen  strlen\n"
"#endif /* __CODECLIB_H__ */\n"
)
